#!/bin/bash



for i in {1..100}; do
	
	ip link del wg0 type wireguard
	ip link add wg0 type wireguard
	ip addr add 10.0.0.12/24 dev wg0
	wg setconf wg0 ./confwgB
	ip link set wg0 up

	sudo dmesg --clear

	ping -c 1 10.0.0.11  >> ping102.txt
	echo $i
	sudo dmesg | grep "CHECKTIME: total" >> durations102.txt
	sudo dmesg | grep "CHECKTIME: decaps" >> decaps102.txt
	sudo dmesg | grep "CHECKTIME: keygen" >> keygen102.txt

done
