#!/bin/bash

sudo dmesg | grep "CHECKTIME: encaps" >> encaps_temp.txt
sudo dmesg | grep "CHECKTIME: searchpeer" >> searchpeer_temp.txt
sudo dmesg --clear

sed -i 's/.*CHECKTIME: encaps \(.*\)/\1/' encaps_temp.txt
echo "Encaps time for $1 peers: " >> measures.txt
awk '{s+=$1; n++} END {if(n) print s/n}' encaps_temp.txt >> measures.txt
echo "\n" >> measures.txt

sed -i 's/.*CHECKTIME: searchpeer \(.*\)/\1/' searchpeer_temp.txt
echo "Search peer time for $1 peers: " >> measures.txt
awk '{s+=$1; n++} END {if(n) print s/n}' searchpeer_temp.txt >> measures.txt
echo "\n" >> measures.txt
