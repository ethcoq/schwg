#!/bin/bash

sudo rmmod wireguard
sudo make -C /usr/src/linux-headers-$(uname -r)/ M=/home/user/newwireguardv2/drivers/net/wireguard/ modules -j$(nproc)
sudo modprobe udp_tunnel
sudo modprobe ip6_udp_tunnel
sudo modprobe curve25519-x86_64
sudo insmod /home/user/newwireguardv2/drivers/net/wireguard/wireguard.ko
echo 'module wireguard +p' | sudo tee /sys/kernel/debug/dynamic_debug/control

ip addr add 192.168.0.11/24 dev enp0s3
ip link add wg0 type wireguard
ip addr add 10.0.0.11/24 dev wg0
wg setconf wg0 ./confwgA100
ip link set wg0 up
